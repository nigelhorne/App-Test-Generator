   63: sub applies_to {
   64: 	my ($self, $doc) = @_;
   65: 
   66: 	# PPI >= 1.270 classifies return as PPI::Statement::Break rather
   67: 	# than the dedicated PPI::Statement::Return class -- scan the whole
   68: 	# document for at least one qualifying return statement. This must
   69: 	# match the document-level pre-filter contract used by
   70: 	# Mutator::generate_mutants (and documented in Mutation::Base) rather
   71: 	# than testing a single node, otherwise every call from
   72: 	# generate_mutants would see $doc itself, which is never a
   73: 	# PPI::Statement::Break, and mutate() would never run.
   74: 	my $returns = $doc->find(sub {
   75: 		my $node = $_[1];
   76: 		return 0 unless $node->isa('PPI::Statement::Break');
   77: 		my $first = $node->schild(0) or return 0;
   78: 		return $first->content eq 'return';
   79: 	}) || [];
   80: 
   81: 	return @{$returns} ? 1 : 0;
   82: }

   84: =head2 mutate
   85: 
   86: Walk a PPI document and generate one mutant for each non-bare return
   87: statement, replacing its expression with C<undef>. For example,
   88: C<return $result> becomes C<return undef>.
   89: 
   90: Bare C<return;> statements are skipped because they already return
   91: undef - mutating them would produce a redundant mutant that can never
   92: be killed.
   93: 
   94:     my $mutation = App::Test::Generator::Mutation::ReturnUndef->new;
   95:     my $doc      = PPI::Document->new(\$source);
   96:     my @mutants  = $mutation->mutate($doc);
   97: 
   98:     for my $m (@mutants) {
   99:         print $m->id, ': ', $m->description, "\n";
  100:     }
  101: 
  102: =head3 Arguments
  103: 
  104: =over 4
  105: 
  106: =item * C<$self>
  107: 
  108: An instance of C<App::Test::Generator::Mutation::ReturnUndef>.
  109: 
  110: =item * C<$doc>
  111: 
  112: A L<PPI::Document> object representing the parsed source to mutate.
  113: The document is not modified by this method.
  114: 
  115: =back
  116: 
  117: =head3 Returns
  118: 
  119: A list of L<App::Test::Generator::Mutant> objects, one per qualifying
  120: return statement found in the document. Returns an empty list if no
  121: non-bare return statements are found.
  122: 
  123: Each mutant carries a C<transform> closure that when called with a
  124: fresh L<PPI::Document> copy will replace the targeted return expression
  125: with the literal C<undef>.
  126: 
  127: =head3 Notes
  128: 
  129: Mutant IDs include both line and column number to ensure uniqueness
  130: when multiple return statements appear in the same source file.
  131: 
  132: Only return statements with an expression child are mutated - bare
  133: C<return;> statements are skipped as they already return undef.
  134: 
  135: Each mutant's optional C<context> field is set to C<conditional> if
  136: the return statement sits inside (or is itself the keyword of) an
  137: C<if>/C<unless>/C<while>/C<until> compound statement, or C<statement>
  138: otherwise; its C<line_content> field holds the raw source text of the
  139: mutated line. Both are consumed by
  140: L<App::Test::Generator::Mutator>'s fast-mode dedup.

  263: # --------------------------------------------------
  264: # Purpose: identify the PPI elements making up the expression
  265: #          being returned by a 'return' statement, excluding
  266: #          the leading 'return' keyword, the trailing statement
  267: #          terminator, and any postfix conditional/loop modifier
  268: #          (if/unless/while/until/for/foreach) and its condition.
  269: # Entry:   a PPI::Statement::Break node already confirmed to be
  270: #          a 'return' statement.
  271: # Exit:    a list of the significant child elements making up
  272: #          the return expression, or an empty list for a bare
  273: #          return (with or without a postfix modifier).
  274: # Side effects: none.
  275: # --------------------------------------------------
  276: sub _return_expr_span {
  277: 	my ($ret) = @_;
  278: 
  279: 	my @children = $ret->schildren;
  280: 	shift @children;
  281: 
  282: 	if(@children && $children[-1]->isa('PPI::Token::Structure') && $children[-1]->content eq ';') {
  283: 		pop @children;
  284: 	}
  285: 
  286: 	for my $i (0 .. $#children) {
  287: 		my $child = $children[$i];
  288: 		next unless $child->isa('PPI::Token::Word');
  289: 		next unless $child->content =~ /^(?:if|unless|while|until|for|foreach)$/;
  290: 		@children = @children[0 .. $i - 1];
  291: 		last;
  292: 	}
  293: 
  294: 	return @children;
  295: }
