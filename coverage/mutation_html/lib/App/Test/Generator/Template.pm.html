 1575: # A documented string or numeric bound sizes the valid data.  Some cases are
 1576: # built from fixed samples rather than from the bound, so drop any that the
 1577: # bound excludes; otherwise a routine that enforces its documented range
 1578: # fails on a value the harness generated as expected-to-live.
 1579: sub _filter_documented_bound_cases
 1580: {
 1581: 	my ($cases, $input) = @_;
 1582: 
 1583: 	my @kept;
 1584: 	CASE:
 1585: 	foreach my $case (@{$cases}) {
 1586: 		# Explicit out-of-bound probes are meant to be run and reported; they
 1587: 		# must survive this filter even though their value is outside a bound.
 1588: 		if($case->{_DOCUMENTED_CONSTRAINTS}) {
 1589: 			push @kept, $case;
 1590: 			next CASE;
 1591: 		}
 1592: 
 1593: 		foreach my $field (keys %{$input}) {
 1594: 			my $spec = $input->{$field};
 1595: 			next unless(ref($spec) eq 'HASH');
 1596: 			next unless _constraint_is_documented($spec, 'min')
 1597: 				|| _constraint_is_documented($spec, 'max');
 1598: 
 1599: 			my $type = $spec->{type} // 'string';
 1600: 			next unless($type =~ /^(?:string|integer|number|float)$/);
 1601: 
 1602: 			my ($value, $found);
 1603: 			if(exists($case->{$field})) {
 1604: 				($value, $found) = ($case->{$field}, 1);
 1605: 			} elsif((ref($case->{'_input'}) eq 'HASH') && exists($case->{'_input'}{$field})) {
 1606: 				($value, $found) = ($case->{'_input'}{$field}, 1);
 1607: 			} elsif(($field eq '_input') && exists($case->{'_input'})) {
 1608: 				($value, $found) = ($case->{'_input'}, 1);
 1609: 			}
 1610: 			next unless $found;
 1611: 
 1612: 			# References, undef and other type probes are not size or range
 1613: 			# values, so the bound does not apply to them.
 1614: 			next unless defined($value) && !ref($value);
 1615: 
 1616: 			if($type eq 'string') {
 1617: 				next CASE unless _string_length_in_documented_bounds($spec, length($value));
 1618: 			} else {
 1619: 				next unless Scalar::Util::looks_like_number($value);
 1620: 				next CASE if _constraint_is_documented($spec, 'min')
 1621: 					&& defined($spec->{min}) && ($value < $spec->{min});
 1622: 				next CASE if _constraint_is_documented($spec, 'max')
 1623: 					&& defined($spec->{max}) && ($value > $spec->{max});
 1624: 			}
 1625: 		}
 1626: 
 1627: 		push @kept, $case;
 1628: 	}
 1629: 
 1630: 	return \@kept;
 1631: }

 1799: 		_DOCUMENTED_CONSTRAINTS => 1,

 1804: # Record a documented-only constraint once per parameter.  Every kind of
 1805: # constraint is reported the same way: bounds, member lists and patterns.

 1807: {
 1808: 	my ($name, $spec) = @_;
 1809: 
 1810: 	return unless ref($spec) eq 'HASH';
 1811: 	my @documented;
 1812: 	push @documented, 'min' if defined($spec->{min}) && _constraint_is_documented($spec, 'min');
 1813: 	push @documented, 'max' if defined($spec->{max}) && _constraint_is_documented($spec, 'max');
 1814: 	push @documented, 'memberof' if _has_memberof($spec) && _constraint_is_documented($spec, 'memberof');

 1824: # Measure the value against the kind of bound the spec declares.  Non-values
 1825: # such as undef, references sent to a scalar parameter and non-numeric strings
 1826: # sent to a numeric parameter are type probes, not bound observations.
 1827: sub _documented_bound_measure
 1828: {
 1829: 	my ($spec, $value) = @_;
 1830: 
 1831: 	my $type = $spec->{type} // 'string';
 1832: 	if($type eq 'string') {
 1833: 		return unless defined($value) && !ref($value);
 1834: 		return (length($value), 'length');
 1835: 	}
 1836: 	if(($type eq 'integer') || ($type eq 'number') || ($type eq 'float')) {
 1837: 		return unless defined($value) && !ref($value) && looks_like_number($value);
 1838: 		return (0 + $value, 'value');
 1839: 	}
 1840: 	if(($type eq 'arrayref') || ($type eq 'array')) {
 1841: 		return unless ref($value) eq 'ARRAY';
 1842: 		return (scalar(@{$value}), 'element count');
 1843: 	}
 1844: 	if($type eq 'hashref') {
 1845: 		return unless ref($value) eq 'HASH';
 1846: 		return (scalar(keys %{$value}), 'key count');
 1847: 	}
 1848: 	return;

 1892: # Split the case's observed values into those inside and outside their POD
 1893: # constraints (bounds, member lists and patterns).  A death is a contradiction

 1895: sub _documented_constraint_findings
 1896: {
 1897: 	my ($input, $schema) = @_;

 1899: 	my @checks;
 1900: 	if((ref($schema) eq 'HASH') && exists($schema->{type}) && !ref($schema->{type})) {
 1901: 		# A single scalar schema: type/min/max are the parameter's spec.
 1902: 		push @checks, ['_input', $schema, $input];

 1907: 			next unless exists($input->{$field});

 1913: 	foreach my $check (@checks) {

 1916: 		my $max_documented = _constraint_is_documented($spec, 'max') && defined($spec->{max});

 1986: 					example => $example,
