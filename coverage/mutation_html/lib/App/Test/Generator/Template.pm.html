 1707: # A documented string or numeric bound sizes the valid data.  Some cases are
 1708: # built from fixed samples rather than from the bound, so drop any that the
 1709: # bound excludes; otherwise a routine that enforces its documented range
 1710: # fails on a value the harness generated as expected-to-live.
 1711: sub _filter_documented_bound_cases
 1712: {
 1713: 	my ($cases, $input) = @_;
 1714: 
 1715: 	my @kept;
 1716: 	CASE:
 1717: 	foreach my $case (@{$cases}) {
 1718: 		# Explicit out-of-bound probes are meant to be run and reported; they
 1719: 		# must survive this filter even though their value is outside a bound.
 1720: 		if($case->{_DOCUMENTED_CONSTRAINTS}) {
 1721: 			push @kept, $case;
 1722: 			next CASE;
 1723: 		}
 1724: 
 1725: 		foreach my $field (keys %{$input}) {
 1726: 			my $spec = $input->{$field};
 1727: 			next unless(ref($spec) eq 'HASH');
 1728: 			next unless _constraint_is_documented($spec, 'min')
 1729: 				|| _constraint_is_documented($spec, 'max');
 1730: 
 1731: 			my $type = $spec->{type} // 'string';
 1732: 			next unless($type =~ /^(?:string|integer|number|float)$/);
 1733: 
 1734: 			my ($value, $found);
 1735: 			if(exists($case->{$field})) {
 1736: 				($value, $found) = ($case->{$field}, 1);
 1737: 			} elsif((ref($case->{'_input'}) eq 'HASH') && exists($case->{'_input'}{$field})) {
 1738: 				($value, $found) = ($case->{'_input'}{$field}, 1);
 1739: 			} elsif(($field eq '_input') && exists($case->{'_input'})) {
 1740: 				($value, $found) = ($case->{'_input'}, 1);
 1741: 			}
 1742: 			next unless $found;
 1743: 
 1744: 			# References, undef and other type probes are not size or range
 1745: 			# values, so the bound does not apply to them.
 1746: 			next unless defined($value) && !ref($value);
 1747: 
 1748: 			if($type eq 'string') {
 1749: 				next CASE unless _string_length_in_documented_bounds($spec, length($value));
 1750: 			} else {
 1751: 				next unless Scalar::Util::looks_like_number($value);
 1752: 				next CASE if _constraint_is_documented($spec, 'min')
 1753: 					&& defined($spec->{min}) && ($value < $spec->{min});
 1754: 				next CASE if _constraint_is_documented($spec, 'max')
 1755: 					&& defined($spec->{max}) && ($value > $spec->{max});
 1756: 			}
 1757: 		}
 1758: 
 1759: 		push @kept, $case;
 1760: 	}
 1761: 
 1762: 	return \@kept;
 1763: }

 1796: {
 1797: 	my ($arg_name, $spec, $mandatory_args) = @_;

 1805: 	return @cases if($spec->{'_fixture'});

 1818: 		);
 1819: 		return @cases;
 1820: 	}
 1821: 
 1822: 	push @cases, { %{$mandatory_args}, ( $arg_name => $outside, _STATUS => 'DIES' ) };
 1823: 
 1824: 	return @cases;

 3129: # What this process's file descriptors point at, as a map of target to the

 3134: sub _open_fds
 3135: {
 3136: 	return unless -d '/proc/self/fd';
 3137: 	opendir(my $dh, '/proc/self/fd') or return;

 3140: 		next if($fd =~ /^\./);

 3142: 		# socket:[12345] and pipe:[12345] differ on every open
 3143: 		$target =~ s/:\[\d+\]$//;
 3144: 		$targets{$target}++;
 3145: 	}
 3146: 	closedir($dh);
 3147:

 3151: # The targets that gained descriptors between two _open_fds() snapshots.
 3152: # The first descriptor ever opened on a target is not a leak: an object that
 3153: # opens its log file or its connection on first use and keeps it is holding a
 3154: # resource, not losing one.  A leak opens the same thing again, so a target
 3155: # that gains a descriptor a second time is reported, as is anything beyond a
 3156: # handful of distinct targets (a new temporary file leaked on every call).
 3157: sub _leaked_fds
 3158: {
 3159: 	my ($before, $after) = @_;
 3160: 
 3161: 	my @leaked;
 3162: 	foreach my $target (sort keys %{$after}) {
 3163: 		my $was = $before->{$target} // 0;
 3164: 		next if($after->{$target} <= $was);
 3165: 		if(($was == 0) && ($after->{$target} == 1) && !$fds_held{$target} && (scalar(keys %fds_held) < 8)) {
 3166: 			$fds_held{$target} = 1;
 3167: 			diag("descriptor on $target opened and kept; taken as held, not leaked") if($ENV{'TEST_VERBOSE'});
 3168: 			next;
 3169: 		}
 3170: 		push @leaked, $target;
 3171: 	}
 3172: 
 3173: 	return @leaked;
