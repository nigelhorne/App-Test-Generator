 1928: 							\%output,
 1929: 							$positions
 1930: 						);
 1931: 					}
 1932: 				}
 1933: 			}
 1934: 		}
 1935: 	}
 1936:

 1960: sub run_test
 1961: {
 1962: 	my($case, $input, $output, $positions) = @_;
 1963: 
 1964: 	if($ENV{'TEST_VERBOSE'}) {
 1965: 		require Data::Dumper;
 1966: 		diag('input: ', Data::Dumper::Dumper($input));
 1967: 	}
 1968: 
 1969: 	my $name = delete local $case->{'_NAME'};
 1970: 	my $properties = delete local $case->{_PROPERTIES};
 1971: 	my $description = delete local $case->{_DESCRIPTION};
 1972: 	my $result;
 1973: 	my $mess;
 1974: 	my @alist = ();
 1975: 	if(defined($input) && !ref($input)) {
 1976: 		# $mess is later used as a sprintf() format string further
 1977: 		# below â€” a literal '%' in $name/$input must be escaped to
 1978: 		# '%%' first, the same as the aggregate branch does for
 1979: 		# $args, or a value like '%s' / '%n' corrupts the sprintf call.
 1980: 		(my $safe_input = $input) =~ s/%/%%/g;
 1981: 		if($name) {
 1982: 			(my $safe_name = $name) =~ s/%/%%/g;
 1983: 			$mess = "[% function %]($safe_name = '$safe_input') %s";
 1984: 		} else {
 1985: 			$mess = "[% function %]('$safe_input') %s";
 1986: 		}
 1987: 		# A scalar _input case is the only argument of a single-parameter
 1988: 		# routine, so it has to be passed as that argument. Without this the
 1989: 		# call below dropped @alist and invoked the routine with no arguments,
 1990: 		# so the case tested nothing while still being reported by name.
 1991: 		$alist[0] = $input if($positions && (scalar(keys %{$positions}) == 1));
 1992: 	} elsif(defined($input)) {
 1993: 		if($positions) {
 1994: 			# Positional args
 1995: 			foreach my $key (keys %{$input}) {
 1996: 				if(($key ne '_STATUS') && ($key ne '_NAME') && ($key ne '_LINE') && ($key ne '_PROPERTIES') && ($key ne '_DESCRIPTION')) {
 1997: 					if(exists($positions->{$key})) {
 1998: 						$alist[$positions->{$key}] = delete $input->{$key};
 1999: 					} else {
 2000: 						diag("Lost position number for $key");
 2001: 					}
 2002: 				}
 2003: 			}
 2004: 			# Keep all slots up to the last defined position, preserving undef
 2005: 			# holes for optional args that were intentionally omitted at lower
 2006: 			# positions.  Stripping ALL undefs (the old grep approach) caused
 2007: 			# a mandatory arg at position N to shift to position N-1 when an
 2008: 			# optional arg at an earlier position was absent â€” the shifted arg
 2009: 			# was then passed to the wrong parameter and the call died.
 2010: 			my $last_def = -1;
 2011: 			for my $i (0..$#alist) { $last_def = $i if defined $alist[$i] }
 2012: 			@alist = $last_def >= 0 ? @alist[0..$last_def] : ();
 2013: 			$input = join(', ', @alist);
 2014: 		} else {
 2015: 			# Named args
 2016: 			if(ref($input) ne 'HASH') {
 2017: 				if($case->{'_STATUS'} ne 'DIES') {
 2018: 					die('Input is missing list of arguments (perhaps you only listed types)');
 2019: 				}
 2020: 				# e.g., we are passing a ref to a scalar to something that only takes a scalar, so it should cause the routine to die
 2021: 				$positions = {};
 2022: 				$alist[0] = $input;
 2023: 			} else {
 2024: 				foreach my $key (sort keys %{$input}) {
 2025: 					if(($key ne '_STATUS') && ($key ne '_NAME') && ($key ne '_LINE') && ($key ne '_PROPERTIES')) {
 2026: 						if(defined($input->{$key})) {
 2027: 							push @alist, "'$key' => '$input->{$key}'";
 2028: 						} else {
 2029: 							push @alist, "'$key' => undef";
 2030: 						}
 2031: 					}
 2032: 				}
 2033: 			}
 2034: 		}
 2035: 		my $args = join(', ', @alist);
 2036: 		$args =~ s/%/%%/g;
 2037: 		$mess = "[% function %]($args) %s";
 2038: 	} else {
 2039: 		$mess = "[% function %] %s";
 2040: 	}
 2041: 
 2042: 	my $status = delete $case->{'_STATUS'} || $output->{'_STATUS'};
 2043: 	my $line = delete $case->{'_LINE'};
 2044: 	my %ENV_before = %ENV;
 2045: 	my $cwd_before = Cwd::getcwd();
 2046: 	my %cwd_files_before = map { $_ => 1 } do { opendir(my $dh, '.') or die $!; readdir($dh) };
 2047: 
 2048: 	local $SIG{ALRM} = sub { die '__TIMEOUT__' };
 2049: 	if((!defined($config{timeout})) || ($config{timeout} > 0)) {
 2050: 		alarm($config{'timeout'} // 10);
 2051: 	}
 2052: 	my $old_warn = $SIG{__WARN__};
 2053: 	my $old_die = $SIG{__DIE__};
 2054: 	my $ok = eval {
 2055: 		if(defined($status)) {
 2056: 			if($status eq 'DIES') {
 2057: 				my $err;
 2058: 				if($positions) {
 2059: 					[% IF position_code %]
 2060: 						if(defined($name)) {
 2061: 							dies_ok { [% position_code %] } sprintf($mess, "dies (position test) - $name (status = DIES)");
 2062: 						} else {
 2063: 							dies_ok { [% position_code %] } sprintf($mess, 'dies (position test, status = DIES)');
 2064: 						}
 2065: 						$err = $@;
 2066: 					[% ELSE %]
 2067: 						ok(0, 'dies: position_code not defined');
 2068: 					[% END %]
 2069: 				} else {
 2070: 					dies_ok { [% call_code %] } sprintf($mess, 'dies');
 2071: 					$err = $@;
 2072: 					ok(!defined($result));
 2073: 				}
 2074: 				ok(defined($err));
 2075: 				ok(length($err));
 2076: 				ok(!ref($err));
 2077: 				if(defined($name)) {
 2078: 					unlike($err, qr/unitialized/, "$name doesn't involve an uninitialized variable");
 2079: 				} else {
 2080: 					unlike($err, qr/unitialized/, "Test doesn't involve an uninitialized variable");
 2081: 				}
 2082: 				return;	# There should be no output to validate
 2083: 			} elsif($status eq 'WARNS') {
 2084: 				warnings_exist { [% call_code %] } qr/./, sprintf($mess, 'warns');
 2085: 			} else {
 2086: 				carp 'unhandled properties in test case; continuing' if(scalar keys %{$properties});
 2087: 				if($positions) {
 2088: 					[% IF position_code %]
 2089: 						if(defined($name)) {
 2090: 							lives_ok { [% position_code %] } sprintf($mess, "survives (position test) - $name (status = LIVES)");
 2091: 						} else {
 2092: 							lives_ok { [% position_code %] } sprintf($mess, 'survives (position test, status = LIVES)');
 2093: 						}
 2094: 					[% ELSE %]
 2095: 						ok(0, 'position_code not defined');
 2096: 					[% END %]
 2097: 				} elsif(defined($description)) {
 2098: 					lives_ok { [% call_code %] } sprintf($mess, "survives ($description; status = LIVES)");
 2099: 				} elsif(defined($line)) {
 2100: 					lives_ok { [% call_code %] } sprintf($mess, "survives (line = $line; status = LIVES)");
 2101: 				} else {
 2102: 					lives_ok { [% call_code %] } sprintf($mess, 'survives (status = LIVES)');
 2103: 				}
 2104: 				if($properties->{idempotent}) {
 2105: 					[% determinism_code %]
 2106: 				}
 2107: 			}
 2108: 		} elsif($positions) {
 2109: 			if(defined($name)) {
 2110: 				lives_ok { [% position_code %] } sprintf($mess, "survives (position test) - $name");
 2111: 				if($properties->{idempotent} && (scalar(@alist) == 1)) {
 2112: 					[% UNLESS module %]
 2113: 						ok([% function %]($alist[0]) eq [% function %]([% function %]($alist[0])), 'function is idempotent');
 2114: 						ok([% function %]($alist[0]) eq [% function %]($alist[0]), 'function is idempotent');

 2116: 				}
 2117: 			} else {
 2118: 				carp 'unhandled properties in test case; continuing' if(scalar keys %{$properties});
 2119: 				lives_ok { [% position_code %] } sprintf($mess, 'survives (position test)');
 2120: 
 2121: 				# An extra argument should be ignored, except for getsetters, so only test if there's more than one arg
 2122: 				if(scalar(@alist) > 1) {
 2123: 					push(@alist, 'foo');
 2124: 					lives_ok { [% position_code %] } sprintf($mess, 'survives (position test, with extra argument)');
 2125: 				}
 2126: 			}
 2127: 			if($properties->{idempotent}) {
 2128: 				[% determinism_code %]
 2129: 			}
 2130: 		} else {
 2131: 			# Status not given, assume set to LIVES
 2132: 			carp 'unhandled properties in test case; continuing' if(scalar keys %{$properties});
 2133: 			if(defined($description)) {
 2134: 				lives_ok { [% call_code %] } sprintf($mess, "survives ($description)");
 2135: 			} elsif(defined($line)) {
 2136: 				lives_ok { [% call_code %] } sprintf($mess, "survives (line = $line)");
 2137: 			} else {
 2138: 				lives_ok { [% call_code %] } sprintf($mess, 'survives');
 2139: 			}
 2140: 			if($properties->{idempotent}) {
 2141: 				[% determinism_code %]
 2142: 			}
 2143: 		}
 2144: 		1;
 2145: 	};
 2146: 
 2147: 	alarm 0;

 2149: 	# Clean up any files the function created in cwd
 2150: 	for my $f (do { opendir(my $dh, '.') or die $!; readdir($dh) }) {
